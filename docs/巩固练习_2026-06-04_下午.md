@pytest.mark.parametrize("city,hobby,username", [
    ("wuhan", "gaming", "张三"),
])
def test_all_actions(test_page, city, hobby, username):
    # 1. 下拉框选武汉
    dropdown = test_page.locator("#city")
    dropdown.select_option("wuhan")
    expect(dropdown).to_have_value("wuhan")
    
    # 2. 勾选游戏
    checkbox = test_page.locator(f"input[value='{hobby}']")
    checkbox.check()
    expect(checkbox).to_be_checked()
    
    # 3. 输入用户名
    user_input = test_page.locator("#username")
    user_input.fill(username)
    expect(user_input).to_have_value(username)

import yaml

with open("data/user.yaml") as f:
    user = yaml.safe_load(f)[0]   # 取第一组

def test_login_variation(test_page):
    # 定位
    username = test_page.locator("#username")
    btn = test_page.locator("#login-btn")
    
    # 操作前断言
    expect(btn).to_be_visible()
    
    # 清空 + 输入
    username.clear()
    username.fill(user["username"])
    
    # 点击 + 断言
    btn.click()
    expect(test_page.locator("#button-result")).to_have_text("✅ 登录按钮被点击了")
